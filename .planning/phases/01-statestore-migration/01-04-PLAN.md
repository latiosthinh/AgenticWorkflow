       - Remove `db`, `reworkCycles`, `eq` imports from `drizzle-orm` and `../db/`.
       - Import `stateStore` from `../state/index.js`.
       - Implement `evaluateCircuitBreaker(workItemId, sourceGate)`:
         ```typescript
         let allowed = true;
         let currentCount = 1;
         await stateStore.updateTicketState(workItemId, (draft) => {
           const prev = draft.reworkCycles?.bounceCount ?? 0;
           currentCount = prev + 1;
           const now = new Date().toISOString();
           if (prev >= 2) {
             allowed = false;
             draft.reworkCycles = {
               bounceCount: currentCount,
               lastBounceAt: now,
               sourceGate,
               escalatedAt: draft.reworkCycles?.escalatedAt ?? now,
               createdAt: draft.reworkCycles?.createdAt ?? now,
               updatedAt: now,
             };
           } else {
             allowed = true;
             draft.reworkCycles = {
               bounceCount: currentCount,
               lastBounceAt: now,
               sourceGate,
               createdAt: draft.reworkCycles?.createdAt ?? now,
               updatedAt: now,
             };
           }
         });
         return { allowed, currentCount };
         ```
       - Implement `resetCircuitBreaker(workItemId)`:
         `await stateStore.updateTicketState(workItemId, (draft) => { if (draft.reworkCycles) { draft.reworkCycles.bounceCount = 0; draft.reworkCycles.escalatedAt = null; draft.reworkCycles.updatedAt = new Date().toISOString(); } });`
