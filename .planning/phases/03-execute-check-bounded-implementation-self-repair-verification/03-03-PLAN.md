   ```typescript
   export async function transitionToDevDone(
     workItemId: number,
     htmlComment: string
   ): Promise<any> {
     const details = await getWorkItemDetails(workItemId);
     const patchDoc = buildDevDonePatch(htmlComment, details.tags);
     return adoClient.updateWorkItem(workItemId, patchDoc);
   }

   export async function flagTicketBlocked(
     workItemId: number,
     htmlComment: string,
     type: 'contract-conflict' | 'repair-exhausted' | 'diff-ceiling'
   ): Promise<any> {
     const details = await getWorkItemDetails(workItemId);
     let patchDoc: any;
     if (type === 'contract-conflict') {
       patchDoc = buildContractConflictPatch(htmlComment, details.tags);
     } else if (type === 'repair-exhausted') {
       patchDoc = buildRepairExhaustedPatch(htmlComment, details.tags);
     } else {
       const tagPatches = buildTagPatch(details.tags, '[diff-ceiling-exceeded]');
       patchDoc = [
         ...tagPatches,
         {
           op: Operation.Replace,
           path: '/fields/System.State',
           value: 'Blocked',
         },
         {
           op: Operation.Add,
           path: '/fields/System.History',
           value: htmlComment,
         },
       ];
     }
     return adoClient.updateWorkItem(workItemId, patchDoc);
   }
   ```

   - In both the fresh 'In Dev' path (when plan has no ambiguities) and resumption path (when developer answered questions and plan is locked), invoke the Phase 3 implementation and verification sequence:
     ```typescript
     // 1. Setup worktree and protect tests
     const { lockedFiles } = protectTestFiles(worktreeResult.worktreePath);
     const git = simpleGit(worktreeResult.worktreePath);

     // 2. In test env or live mode, perform bounded code editing
     // ...

     // 3. Guard diff ceiling (<250 LOC)
     const diffStat = await calculateCumulativeDiff(git, worktreeResult.baseCommit);
     if (diffStat.totalLoc > 250) {
       const comment = formatL3EvidenceComment({ ... }); // or diff ceiling diagnostic
       await flagTicketBlocked(workItemId, comment, 'diff-ceiling');
       return;
     }

     // 4. Guard test assertion immutability
     const rawDiff = await git.raw(['diff', '--name-status', worktreeResult.baseCommit]);
     const immutabilityResult = checkTestImmutability(rawDiff, lockedFiles);
     if (!immutabilityResult.valid) {
       await flagTicketBlocked(workItemId, `<h3>[Contract Conflict] Protected test files modified: ${immutabilityResult.violations.join(', ')}</h3>`, 'contract-conflict');
       return;
     }

     // 5. Execute local tests and self-repair loop
     const repairResult = await executeRepairLoop({
       worktreePath: worktreeResult.worktreePath,
       git,
       workItemId,
     });

     if (!repairResult.success) {
       await flagTicketBlocked(
         workItemId,
         `<h3>[Repair Exhausted] Test self-repair budget exhausted</h3><p>WIP branch created: <code>${repairResult.wipBranch}</code></p><pre>${repairResult.diagnostics}</pre>`,
         'repair-exhausted'
       );
       return;
     }

     // 6. Record L3 evidence and transition to Dev Done
     await recordL3Evidence({
       workItemId,
       revId,

       passed: 1,
       failed: 0,

       failed: 0,
       durationMs: 100,
       gitDiffStat: diffStat.rawStat || 'clean',
     });

     await transitionToDevDone(workItemId, comment);
     await cleanupWorktree(process.cwd(), worktreeResult.worktreePath);
     ```
